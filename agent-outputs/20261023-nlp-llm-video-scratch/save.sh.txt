#!/usr/bin/env bash
# Commit and push the builders' current work plus fresh copies of their
# reports and progress logs. Usage: save.sh "<commit subject>" ["<body>"]
set -e
S=/tmp/claude-0/-home-user-dataslope/2a3cc9a3-7547-5d18-88e1-2b54dd0db0d9/scratchpad

  for d in video/src/videos/$f video/public/images/$f video/public/narration/$f video/takes/$f; do [ -e $d ] && P="$P $d"; done
done

BODY="${2:-The latest files, reports and progress logs from the builders.}"

$1

$BODY

Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>
Claude-Session: https://claude.ai/code/session_011WWHFvWR4omEuyWqKc1KJG

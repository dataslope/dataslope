Every lesson is in there: `box-sizing` at the top, custom properties
for the theme, a flex nav with the `margin-right: auto` split and
`position: sticky`, a grid-centered hero with `clamp()` type, an
`auto-fit` feature grid that reflows with no media query, and hover
transitions on the buttons and cards. Change `--brand` once and the
whole page re-themes.

  gap: 1rem;
  padding: 1rem;
  background: #0f172a;
  color: white;
}
.brand { font-weight: 800; margin-right: auto; }
.links { display: flex; gap: 1rem; }
.links a { color: #cbd5e1; text-decoration: none; }

  .nav {

    align-items: flex-start;
  }

`,

  tests={[
    {
      id: "has_query",
      name: "There is a max-width media query",
      code: `let found = false;
for (const sheet of Array.from(document.styleSheets)) {
  let rules;
  try { rules = sheet.cssRules; } catch (e) { continue; }
  for (const rule of Array.from(rules || [])) {
    if (rule.type === 4 && /max-width/i.test(rule.conditionText || rule.media?.mediaText || "")) found = true;
  }
}
if (!found) throw new Error("Add a @media (max-width: 500px) query.");`,
    },
    {
      id: "stacks",
      name: "The query stacks the nav into a column",
      code: `let ok = false;
for (const sheet of Array.from(document.styleSheets)) {
  let rules;
  try { rules = sheet.cssRules; } catch (e) { continue; }
  for (const rule of Array.from(rules || [])) {
    if (rule.type !== 4) continue;
    for (const inner of Array.from(rule.cssRules || [])) {
      const css = inner.style?.cssText || inner.cssText || "";
      if (/\\.nav/.test(inner.selectorText || "") && /flex-direction:\\s*column/.test(css)) ok = true;
    }
  }
}
if (!ok) throw new Error("Inside the media query, set .nav { flex-direction: column }.");`,
    },

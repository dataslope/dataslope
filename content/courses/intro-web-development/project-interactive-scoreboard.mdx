Notice the division of labour: the HTML never mentions colors, the
CSS never mentions clicks, the JavaScript never invents markup, each
file could be handed to a different person.

## Your turn

The challenge version below has the HTML finished, most of the CSS
finished, and an empty script. Wire it up, and add one new feature
the reference doesn't have: a **reset button**.

    },
    {
      filename: "script.js",
      starterCode: `// Keep each team's score in a variable, update the matching
// element on click, and reset both to 0 on #reset.
`,
      solutionCode: `var home = 0;
var away = 0;

var homeEl = document.querySelector("#score-home");
var awayEl = document.querySelector("#score-away");

document.querySelector("#add-home").addEventListener("click", function () {
  home = home + 1;
  homeEl.textContent = home;
});

document.querySelector("#add-away").addEventListener("click", function () {
  away = away + 1;
  awayEl.textContent = away;
});

document.querySelector("#reset").addEventListener("click", function () {
  home = 0;
  away = 0;
  homeEl.textContent = 0;
  awayEl.textContent = 0;
});
`,
    },
  ]}
  tests={[
    {
      id: "home",
      name: "+1 Home raises the home score",
      code: `document.querySelector("#add-home").click();
await new Promise((r) => setTimeout(r, 80));
document.querySelector("#add-home").click();
await new Promise((r) => setTimeout(r, 30));
const text = (document.querySelector("#score-home").textContent || "").trim();
if (text !== "2") {
  throw new Error("After two Home clicks the home score should read '2', got '" + text + "'.");
}`,
    },
    {
      id: "away",
      name: "+1 Away raises the away score independently",
      code: `document.querySelector("#add-away").click();
await new Promise((r) => setTimeout(r, 30));
const away = (document.querySelector("#score-away").textContent || "").trim();
const home = (document.querySelector("#score-home").textContent || "").trim();
if (away !== "1") {
  throw new Error("After one Away click the away score should read '1', got '" + away + "'.");
}
if (home !== "2") {
  throw new Error("Clicking Away must not change the home score (expected it to stay '2', got '" + home + "').");
}`,
    },
    {
      id: "reset",
      name: "Reset returns both scores to 0",
      code: `document.querySelector("#reset").click();
await new Promise((r) => setTimeout(r, 30));
const home = (document.querySelector("#score-home").textContent || "").trim();
const away = (document.querySelector("#score-away").textContent || "").trim();
if (home !== "0" || away !== "0") {
  throw new Error("Reset should zero both scores, got home='" + home + "', away='" + away + "'.");
}`,
    },
    {
      id: "reset_then_add",
      name: "Counting still works after a reset",
      code: `document.querySelector("#add-home").click();
await new Promise((r) => setTimeout(r, 30));
const home = (document.querySelector("#score-home").textContent || "").trim();
if (home !== "1") {
  throw new Error("After reset + one Home click the score should read '1', got '" + home + "', remember to reset the VARIABLES, not just the text.");
}`,
    },
  ]}
/>

The last test is the sneaky one: if you reset only the *displayed
text* but not the score *variables*, the next click jumps back to the
old count. State lives in your variables; the DOM just displays it,
a distinction that will matter again in the React lesson.

## 2. LINQ basics

<ChallengeCard
  adapter="csharp"
  title="Sum of squares"
  instructions={`Given the integer array \`nums\` below, compute the sum of squares of its elements and print the result.`}
  files={[
    {
      filename: "main.cs",
      starterCode: `using System;
using System.Linq;

class Program
{
    static void Main()
    {
        int[] nums = { 1, 2, 3, 4, 5 };
        // your code here
    }
}
`,
      solutionCode: `using System;

class Program
{
    static void Main()
    {
        int[] nums = { 1, 2, 3, 4, 5 };
        int sum = 0;
        foreach (int n in nums)
            sum += n * n;
        Console.WriteLine(sum);
    }
}
`,
    },
  ]}
  tests={[
    {
      id: "sum_squares",
      name: "Sum of squares is 55",
      description: "1+4+9+16+25 = 55",
      expect: { stdoutEquals: "55" },
    },
  ]}
/>

## 3. Class with property

<ChallengeCard
  adapter="csharp"
  title="Implement a Point's distance"
  instructions={`Add a method \`DistanceTo(Point other)\` to the \`Point\` class that returns the Euclidean distance between two points. The driver code prints the distance from (0, 0) to (3, 4), which should be exactly \`5\`.`}
  files={[
    {
      filename: "main.cs",
      starterCode: `using System;

class Point
{
    public double X { get; }
    public double Y { get; }

    public Point(double x, double y)
    {
        X = x;
        Y = y;
    }

    // your code here
}

class Program
{
    static void Main()
    {
        var a = new Point(0, 0);
        var b = new Point(3, 4);
        Console.WriteLine(a.DistanceTo(b));
    }
}
`,
      solutionCode: `using System;

class Point
{
    public double X, Y;
    public Point(double x, double y)
    {
        X = x;
        Y = y;
    }
    public double DistanceTo(Point other)
    {
        double dx = X - other.X, dy = Y - other.Y;
        return Math.Sqrt(dx * dx + dy * dy);
    }
}

class Program
{
    static void Main()
    {
        var a = new Point(0, 0);
        var b = new Point(3, 4);
        Console.WriteLine(a.DistanceTo(b));
    }
}
`,
    },
  ]}
  tests={[
    {
      id: "distance",
      name: "Distance is 5",
      description: "DistanceTo((3, 4)) from origin",
      expect: { stdoutEquals: "5" },
    },
  ]}
/>

## Multi-file challenge

Multi-file challenges open with a non-sortable, non-closeable tab bar
above the editor. Every `.cs` file is staged into Roslyn's
compilation unit before each run, so the entry script can reference
types defined in sibling files.

<ChallengeCard
  adapter="csharp"
  title="Implement Greeter.Greet()"
  instructions={`Open \`Greeter.cs\` and implement \`Greet()\` so that it returns the string \`"Hello, World!"\`. The entry script \`Main.cs\` already constructs a \`Greeter\` and prints the result.`}
  entryFilename="Main.cs"
  files={[
    {
      filename: "Main.cs",
      starterCode: `var g = new Greeter();\nSystem.Console.WriteLine(g.Greet());\n`,
    },
    {
      filename: "Greeter.cs",
      starterCode: `public class Greeter\n{\n    public string Greet()\n    {\n        // your code here\n        return "";\n    }\n}\n`,
      solutionCode: `public class Greeter\n{\n    public string Greet()\n    {\n        return "Hello, World!";\n    }\n}\n`,
    },
  ]}
  tests={[
    {
      id: "greets",
      name: "Prints `Hello, World!`",
      description: "Exact stdout match",
      expect: { stdoutEquals: "Hello, World!" },
    },
  ]}
/>

## Reading standard input

A card takes the same `stdin` prop. The panel is staged as `stdin.txt`
and read by `Console.ReadLine`, and the tests grade output produced
from it.

<ChallengeCard
  adapter="csharp"
  title="Total an order"
  stdin={`widget,2,3.50
gizmo,1,12.00
sprocket,4,0.75`}
  instructions={`Each line of input is \`name,quantity,unitPrice\`. Print the order total,
formatted to two decimals:

\`\`\`text
Total: $22.00
\`\`\``}
  files={[
    {
      filename: "Program.cs",
      starterCode: `using System;
using System.Globalization;

// read each line, multiply quantity by unit price, print the total
`,
      solutionCode: `using System;
using System.Globalization;

decimal total = 0m;
string? line;
while ((line = Console.ReadLine()) != null)
{
    var parts = line.Split(',');
    if (parts.Length != 3) continue;
    var qty = int.Parse(parts[1], CultureInfo.InvariantCulture);
    var unit = decimal.Parse(parts[2], CultureInfo.InvariantCulture);
    total += qty * unit;
}

Console.WriteLine("Total: $" + total.ToString("0.00", CultureInfo.InvariantCulture));
`,
    },
  ]}
  tests={[
    {
      id: "total",
      name: "Prints the order total",
      description: "Exact stdout match (trimmed)",
      expect: { stdoutEquals: "Total: $22.00" },
    },
  ]}
/>

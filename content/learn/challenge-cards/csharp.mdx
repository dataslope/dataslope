## 3. Class with property

<ChallengeCard
  adapter="csharp"
  title="Implement a Point's distance"
  category="classes · properties"
  estimatedTime="~4 min"
  instructions={<>
    <p>
      Add a method <code>DistanceTo(Point other)</code> to the{' '}
      <code>Point</code> class that returns the Euclidean distance
      between two points. The driver code prints the distance from{' '}
      (0, 0) to (3, 4), which should be exactly <code>5</code>.
    </p>
  </>}
  initialCode={`using System;

class Point
{
    public double X { get; }
    public double Y { get; }

    public Point(double x, double y)
    {
        X = x;
        Y = y;
    }

    // your code here
}

class Program
{
    static void Main()
    {
        var a = new Point(0, 0);
        var b = new Point(3, 4);
        Console.WriteLine(a.DistanceTo(b));
    }
}
`}
  solutionCode={`using System;

class Point {
    public double X, Y;
    public Point(double x, double y) { X = x; Y = y; }
    public double DistanceTo(Point other) {
        double dx = X - other.X, dy = Y - other.Y;
        return Math.Sqrt(dx * dx + dy * dy);
    }
}

class Program {
    static void Main() {
        var a = new Point(0, 0);
        var b = new Point(3, 4);
        Console.WriteLine(a.DistanceTo(b));
    }
}
`}
  tests={[
    {
      id: "distance",
      name: "Distance is 5",
      description: "DistanceTo((3, 4)) from origin",
      expect: { stdoutEquals: "5" },
    },
  ]}
/>
